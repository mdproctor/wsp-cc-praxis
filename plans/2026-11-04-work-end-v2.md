# Work-End v2: Atomic Close Ceremony — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use
> subagent-driven-development (recommended) or executing-plans to
> implement this plan task-by-task. Each task follows TDD

**Focal issue:** #499 — Redesign work-end: atomic ceremony with separated queue
**Issue group:** #499, #498

**Goal:** Replace the 10,800-LOC work-end orchestrator system with a ~600-LOC
atomic close ceremony, separating the issue queue from close-ceremony state.

**Architecture:** Three-phase model (pre-close → ceremony → post-close) with
hard isolation boundaries. The ceremony is a single Python invocation that
either succeeds or fails cleanly. The issue queue (`.queue`) is separated
from ephemeral ceremony state (`.close-state`). See spec at
`specs/issue-499-redesign-work-end/2026-11-04-work-end-v2-design.md`.

**Tech Stack:** Python 3.14, pytest, git CLI

**Import convention:** The `work-end/` directory uses a hyphen on disk
but Python imports use `work_end` (underscore). The directory already has
`__init__.py` and existing imports use `from work_end.X import`. Follow
this convention for `close.py`.

## Global Constraints

- Queue file format uses em dash (`—`) separator, not colon
- Status values: `active` | `paused` | `drained` | `idle` — no others
- No `remove_queue()` function — the queue is never deleted
- `auto_action="remove_plan"` patterns must not survive in any form
- All ceremony steps must be idempotent
- IntelliJ MCP required for code navigation during implementation

```python
# project/queue_io.py
from __future__ import annotations

import re
from dataclasses import dataclass, field
from pathlib import Path


@dataclass
class QueueItem:
    repo: str
    number: int
    title: str
    completed: bool
    active: bool

    @property
    def ref(self) -> str:
        return f"{self.repo}#{self.number}"


@dataclass
class QueueState:
    branch: str = ""
    issue_repo: str = ""
    status: str = "idle"


@dataclass
class Queue:
    items: list[QueueItem] = field(default_factory=list)
    state: QueueState = field(default_factory=QueueState)
    deferred_lines: list[str] = field(default_factory=list)
    tail_lines: list[str] = field(default_factory=list)

    def active_item(self) -> QueueItem | None:
        for item in self.items:
            if item.active:
                return item
        return None

    def complete_active(self) -> None:
        item = self.active_item()
        if item:
            item.completed = True
            item.active = False

    def has_next(self) -> bool:
        for item in self.items:
            if not item.completed and not item.active:
                return True
        return False

    def activate_next(self) -> None:
        for item in self.items:
            if not item.completed and not item.active:
                item.active = True
                return

    def set_status(self, status: str) -> None:
        if status not in ("active", "paused", "drained", "idle"):
            raise ValueError(f"invalid status: {status}")
        self.state.status = status


_ITEM_RE = re.compile(
    r"^- \[([ x])\] (.+?)(?:\s+← active)?$"
)
_REF_RE = re.compile(
    r"^(?:([A-Za-z0-9._-]+/[A-Za-z0-9._-]+))?#(\d+)\s*—\s*(.+)$"
)


def _parse_item(line: str) -> QueueItem | None:
    m = _ITEM_RE.match(line.strip())
    if not m:
        return None
    completed = m.group(1) == "x"
    active = "← active" in line
    rest = m.group(2).strip()
    if "← active" in rest:
        rest = rest.replace("← active", "").strip()
    rm = _REF_RE.match(rest)
    if rm:
        return QueueItem(repo=rm.group(1) or "", number=int(rm.group(2)),
                         title=rm.group(3).strip(),
                         completed=completed, active=active)
    return None


def read_queue(workspace: Path) -> Queue | None:
    queue_path = workspace / ".queue"
    plan_path = workspace / ".plan"

    if queue_path.exists():
        path = queue_path
    elif plan_path.exists():
        migrate_plan_to_queue(workspace)
        if queue_path.exists():
            path = queue_path
        else:
            return None
    else:
        return None

    content = path.read_text()
    lines = content.splitlines()

    items: list[QueueItem] = []
    state = QueueState()
    deferred: list[str] = []
    tail: list[str] = []
    section = "queue"

    for line in lines:
        stripped = line.strip()
        if stripped == "## State":
            section = "state"
            continue
        elif stripped == "## Deferred":
            section = "deferred"
            continue
        elif stripped.startswith("## "):
            section = "tail"

        if section == "queue":
            item = _parse_item(line)
            if item:
                items.append(item)
        elif section == "state":
            if ":" in stripped:
                key, val = stripped.split(":", 1)
                key, val = key.strip(), val.strip()
                if key == "branch":
                    state.branch = val
                elif key == "issue-repo":
                    state.issue_repo = val
                elif key == "status":
                    state.status = val
                elif key == "state":
                    state.status = _migrate_lifecycle_state(val)
        elif section == "deferred":
            deferred.append(line)
        elif section == "tail":
            tail.append(line)

    return Queue(items=items, state=state, deferred_lines=deferred,
                 tail_lines=tail)


def write_queue(workspace: Path, queue: Queue) -> None:
    lines: list[str] = []
    for item in queue.items:
        check = "x" if item.completed else " "
        marker = "        ← active" if item.active else ""
        if item.title:
            lines.append(f"- [{check}] {item.ref} — {item.title}{marker}")
        else:
            lines.append(f"- [{check}] {item.ref}{marker}")
    lines.append("")
    lines.append("## State")
    lines.append(f"branch: {queue.state.branch}")
    lines.append(f"issue-repo: {queue.state.issue_repo}")
    lines.append(f"status: {queue.state.status}")
    if queue.deferred_lines:
        lines.append("")
        lines.append("## Deferred")
        lines.extend(queue.deferred_lines)
    if queue.tail_lines:
        lines.extend(queue.tail_lines)
    lines.append("")
    (workspace / ".queue").write_text("\n".join(lines))


def _migrate_lifecycle_state(state_val: str) -> str:
    if state_val.startswith("closing:"):
        return "active"
    mapping = {
        "active": "active",
        "scaffolded": "active",
        "transitioning": "active",
        "paused": "paused",
        "drained": "drained",
        "idle": "idle",
    }
    return mapping.get(state_val, "active")


def migrate_plan_to_queue(workspace: Path) -> None:
    plan_path = workspace / ".plan"
    queue_path = workspace / ".queue"
    if not plan_path.exists() or queue_path.exists():
        return
    content = plan_path.read_text()
    content = re.sub(r"^state: .+$", "", content, flags=re.MULTILINE)
    content = content.replace("evidence-era: true\n", "")
    lines = content.splitlines()
    out: list[str] = []
    in_state = False
    status_written = False
    for line in lines:
        if line.strip() == "## State":
            in_state = True
            out.append(line)
            continue
        if in_state and line.strip().startswith("## "):
            in_state = False
        if in_state and not line.strip() and not status_written:
            continue
        if in_state and "status:" not in content and not status_written:
            out.append("status: active")
            status_written = True
        out.append(line)
    if not status_written:
        has_state_section = any(l.strip() == "## State" for l in out)
        if has_state_section:
            idx = next(i for i, l in enumerate(out) if l.strip() == "## State")
            out.insert(idx + 1, "status: active")
    queue_path.write_text("\n".join(out) + "\n")
    plan_path.rename(plan_path.with_suffix(".plan.bak"))
```

```python
@dataclass
class Finding:
    scenario: str
    severity: str
    detail: str

def validate_queue(workspace: Path, queue: Queue,
                   current_branch: str, base_branch: str) -> list[Finding]:
    findings = []
    if not queue.state.status:
        findings.append(Finding("S1", "high", "queue has no status"))
    if queue.state.branch and current_branch != base_branch:
        if current_branch != queue.state.branch:
            findings.append(Finding("S5", "high",
                f"branch mismatch: queue={queue.state.branch} actual={current_branch}"))
    queue_path = workspace / ".queue"
    if queue_path.exists():
        content = queue_path.read_text()
        if "<<<<<<" in content or "======" in content or ">>>>>>" in content:
            findings.append(Finding("S13", "critical", "merge conflict markers in .queue"))
    return findings
```

```python
def discover_repo_topology(workspace: Path) -> list[tuple]:
    """Return list of (repo, base_branch, transport, canonical, push_target, mirror_target).

    Slot detection: import from slot_core.py (get_slot_repos,
    resolve_canonical_repo) rather than reimplementing.
    Non-slot: discover project repo (readlink proj/) + workspace.
    Fork model: check for 'upstream' remote → push_target='upstream', mirror='origin'.
    Base branch: read from repo config, default 'main'.
    """
    from slot_core import get_slot_repos, resolve_canonical_repo

    results = []
    slot_repos = get_slot_repos(workspace)
    if slot_repos:
        for repo in slot_repos:
            canonical = resolve_canonical_repo(repo)
            base = _get_base_branch(repo)
            push, mirror = _detect_fork_model(repo)
            results.append((repo, base, "two-hop", canonical, push, mirror))
    else:
        project = _resolve_project(workspace)
        if project:
            base = _get_base_branch(project)
            push, mirror = _detect_fork_model(project)
            results.append((project, base, "direct", None, push, mirror))
        base = _get_base_branch(workspace)
        push, mirror = _detect_fork_model(workspace)
        results.append((workspace, base, "direct", None, push, mirror))
    return results

def _detect_fork_model(repo: Path) -> tuple[str, str]:
    """Check for upstream remote (fork model)."""
    result = subprocess.run(
        ["git", "-C", str(repo), "remote"],
        capture_output=True, text=True)
    remotes = result.stdout.strip().split("\n")
    if "upstream" in remotes:
        return ("upstream", "origin")
    return ("origin", "")
```
